289bool CFrmChineseChess::ConvertCoordinate(long *x, long *y, int *i, int *j, ENUM_ConvertCoordinate eCC)
290{
291 switch (eCC)
292 {
293 case XYToIJ:
294 *i = (*x - m_QiPangStartX + m_QiPangDistance / 2) / m_QiPangDistance;
295 *j = (*y - m_QiPangStartY + m_QiPangDistance / 2) / m_QiPangDistance;
296 break;
297 case IJToXY:
298 *x = m_QiPangStartX + *i * m_QiPangDistance;
299 *y = m_QiPangStartY + *j * m_QiPangDistance;
300 break;
301 }
302 return true;
303}

313 if (width > height)

321 m_QiPangDistance = width / 10;

324 }

473void CFrmChineseChess::DrawXinWei(QPainter *pPainter, int i, int j, ENUM_XINWEI xinwei)
474{
475 int d, ds;
476 long x, y;
477
478 d = m_QiPangDistance * 4 / 10;
479 ds = m_QiPangDistance * 2 / 10;
480
481 ConvertCoordinate(&x, &y, &i, &j, IJToXY);
482
483 QPen pen;
484 pen.setColor(m_QiPangColor);
485 pen.setWidth(1);
486 pPainter->setPen(pen);
487
488 if (xinwei != Left_XinWei)
489 {
490 pPainter->drawLine(x + ds, y + ds, x + d, y + ds);
491 pPainter->drawLine(x + ds, y - ds, x + d, y - ds);
492
493 pPainter->drawLine(x + ds, y + ds, x + ds, y + d);
494
495 pPainter->drawLine(x + ds, y - ds, x + ds, y - d);
496 }
497
498 if (xinwei != Right_XinWei)
499 {
500 pPainter->drawLine(x - ds, y - ds, x - ds, y - d);
501
502 pPainter->drawLine(x - ds, y + ds, x - ds, y + d);

504 pPainter->drawLine(x - ds, y + ds, x - d, y + ds);
505
506 pPainter->drawLine(x - ds, y - ds, x - d, y - ds);
